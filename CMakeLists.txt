cmake_minimum_required(VERSION 3.28)
project(MultiCam LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
# Capture/conversion must remain optimized even with empty cached Release flags.
# Apply to the capture library too, before entering its subdirectory.
add_compile_options("$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Release>:/Ob2>")
add_compile_definitions("$<$<CONFIG:Release>:NDEBUG>")
add_subdirectory(third_party/libdshowcapture)
# Generic USB SDR capture does not need vendor HDR/encoder controls or ATL.
get_target_property(capture_sources libdshowcapture SOURCES)
list(FILTER capture_sources EXCLUDE REGEX "^external/|^source/device-vendor\\.cpp$")
set_property(TARGET libdshowcapture PROPERTY SOURCES "${capture_sources}")
target_sources(libdshowcapture PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/native/vendor-sdr.cpp)
target_compile_options(libdshowcapture PRIVATE /EHsc)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/utf-8 /W4 /EHsc)
add_executable(MultiCam.Worker native/worker.cpp)
target_include_directories(MultiCam.Worker PRIVATE third_party/libdshowcapture)
target_link_libraries(MultiCam.Worker PRIVATE libdshowcapture ole32 oleaut32 advapi32 cfgmgr32 setupapi)
add_library(MultiCam.Camera SHARED native/camera.cpp native/camera.def)
target_include_directories(MultiCam.Camera PRIVATE third_party/libdshowcapture)
target_link_libraries(MultiCam.Camera PRIVATE libdshowcapture ole32 oleaut32 advapi32 winmm)
add_executable(MultiCam.Integration native/integration.cpp)
target_include_directories(MultiCam.Integration PRIVATE third_party/libdshowcapture)
target_link_libraries(MultiCam.Integration PRIVATE libdshowcapture ole32 oleaut32 advapi32 winmm)
add_executable(MultiCam.JpegTest EXCLUDE_FROM_ALL tests/jpeg.cpp)
add_executable(MultiCam.ViewerCoreTest EXCLUDE_FROM_ALL tests/viewer_core.cpp)
add_executable(MultiCam.PipelineTest EXCLUDE_FROM_ALL tests/pipeline_test.cpp)
# HLSL is compiled at build time with fxc; no runtime d3dcompiler dependency.
find_program(FXC_EXE fxc HINTS "$ENV{WindowsSdkVerBinPath}/x64" "$ENV{WindowsSdkBinPath}/x64")
if(NOT FXC_EXE)
  file(GLOB fxc_candidates "$ENV{ProgramFiles\(x86\)}/Windows Kits/10/bin/*/x64/fxc.exe")
  list(SORT fxc_candidates ORDER DESCENDING)
  if(fxc_candidates)
    list(GET fxc_candidates 0 FXC_EXE)
  endif()
endif()
set(VIEWER_SHADER_DIR ${CMAKE_CURRENT_BINARY_DIR}/shaders)
function(add_hlsl_header out_list entry profile)
  set(header ${VIEWER_SHADER_DIR}/${entry}.h)
  add_custom_command(OUTPUT ${header}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${VIEWER_SHADER_DIR}
    COMMAND ${FXC_EXE} /nologo /T ${profile} /E ${entry} /Vn g_${entry} /Fh ${header} ${CMAKE_CURRENT_SOURCE_DIR}/native/viewer.hlsl
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/native/viewer.hlsl VERBATIM)
  set(${out_list} ${${out_list}} ${header} PARENT_SCOPE)
endfunction()
if(FXC_EXE)
  set(VIEWER_SHADERS "")
  add_hlsl_header(VIEWER_SHADERS vs_main vs_4_0)
  add_hlsl_header(VIEWER_SHADERS ps_convert ps_4_0)
  add_hlsl_header(VIEWER_SHADERS ps_blit ps_4_0)
  add_executable(MultiCam.RendererTest EXCLUDE_FROM_ALL tests/renderer_test.cpp native/renderer.cpp ${VIEWER_SHADERS})
  target_include_directories(MultiCam.RendererTest PRIVATE ${VIEWER_SHADER_DIR})
  target_link_libraries(MultiCam.RendererTest PRIVATE d3d11 dxgi)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_executable(MultiCam.Viewer WIN32 native/viewer.cpp native/renderer.cpp ${VIEWER_SHADERS})
    target_include_directories(MultiCam.Viewer PRIVATE ${VIEWER_SHADER_DIR})
    target_link_libraries(MultiCam.Viewer PRIVATE d3d11 dxgi user32 gdi32 shell32 winmm)
  endif()
else()
  message(WARNING "fxc.exe not found; viewer targets are skipped")
endif()
