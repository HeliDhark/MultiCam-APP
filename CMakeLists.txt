cmake_minimum_required(VERSION 3.28)
project(MultiCam LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
# Capture/conversion must remain optimized even with empty cached Release flags.
# Apply to the capture library too, before entering its subdirectory.
add_compile_options("$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Release>:/Ob2>")
add_compile_definitions("$<$<CONFIG:Release>:NDEBUG>")
add_subdirectory(third_party/libdshowcapture)
# Generic USB SDR capture does not need vendor HDR/encoder controls or ATL.
get_target_property(capture_sources libdshowcapture SOURCES)
list(FILTER capture_sources EXCLUDE REGEX "^external/|^source/device-vendor\\.cpp$")
set_property(TARGET libdshowcapture PROPERTY SOURCES "${capture_sources}")
target_sources(libdshowcapture PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/native/vendor-sdr.cpp)
target_compile_options(libdshowcapture PRIVATE /EHsc)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/utf-8 /W4 /EHsc)
add_executable(MultiCam.Worker native/worker.cpp)
target_include_directories(MultiCam.Worker PRIVATE third_party/libdshowcapture)
target_link_libraries(MultiCam.Worker PRIVATE libdshowcapture ole32 oleaut32 advapi32 cfgmgr32 setupapi)
add_library(MultiCam.Camera SHARED native/camera.cpp native/camera.def)
target_include_directories(MultiCam.Camera PRIVATE third_party/libdshowcapture)
target_link_libraries(MultiCam.Camera PRIVATE libdshowcapture ole32 oleaut32 advapi32 winmm)
add_executable(MultiCam.Integration native/integration.cpp)
target_include_directories(MultiCam.Integration PRIVATE third_party/libdshowcapture)
target_link_libraries(MultiCam.Integration PRIVATE libdshowcapture ole32 oleaut32 advapi32 winmm)
add_executable(MultiCam.JpegTest EXCLUDE_FROM_ALL tests/jpeg.cpp)
